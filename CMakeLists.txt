cmake_minimum_required(VERSION 3.14)

project(ThorVG
    VERSION 1.0.0
    DESCRIPTION "Thor Vector Graphics Library"
    LANGUAGES CXX
)

# --- Options de configuration ---
option(TVG_RENDERER_SW "Activer le rendu logiciel (CPU)" ON)
option(TVG_LOADER_SVG  "Activer le support SVG" ON)
option(TVG_LOADER_LOTTIE "Activer le support Lottie" OFF)
option(TVG_THREAD "Activer le support du multi-threading" ON)

# --- GENERATION DE CONFIG.H (Le correctif est ici) ---
# On crée le fichier config.h dans le dossier de build (binaire)
# car les sources s'attendent à pouvoir faire #include "config.h"
set(GENERATED_CONFIG_H "${CMAKE_CURRENT_BINARY_DIR}/config.h")


file(WRITE ${GENERATED_CONFIG_H} 
"/* Auto-generated by CMake for ThorVG */
#ifndef TVG_CONFIG_H
#define TVG_CONFIG_H

#define TVG_VERSION \"${PROJECT_VERSION}\"
#define THORVG_VERSION_STRING \"1.0.0\"

#endif // TVG_CONFIG_H
")

# --- Liste des sources ---
set(TVG_SOURCES)
set(TVG_INCLUDE_DIRS "inc")
set(TVG_PRIVATE_INCLUDE_DIRS 
  "src/common" 
  "src/renderer" 
  "src/renderer/sw_engine"
  "src/loaders/svg"
  "src/loaders/lottie"
  "src/loaders/raw"
)

# 1. Coeur de la librairie (Common & API)
file(GLOB_RECURSE TVG_COMMON_SRC "src/common/*.cpp" "src/renderer/*.cpp" "src/loaders/raw/*.cpp")

# Exclusions des dossiers spécifiques
list(FILTER TVG_COMMON_SRC EXCLUDE REGEX "src/renderer/gl_engine/.*")
list(FILTER TVG_COMMON_SRC EXCLUDE REGEX "src/renderer/sw_engine/.*")
list(FILTER TVG_COMMON_SRC EXCLUDE REGEX "src/renderer/wg_engine/.*")
list(APPEND TVG_SOURCES ${TVG_COMMON_SRC})

add_compile_definitions(THORVG_FILE_IO_SUPPORT)

# 2. Renderer Software (Rasterizer CPU)
if(TVG_RENDERER_SW)
    file(GLOB_RECURSE TVG_SW_SRC "src/renderer/sw_engine/*.cpp")
    list(APPEND TVG_SOURCES ${TVG_SW_SRC})
    file(GLOB_RECURSE TVG_SW_HDR "src/renderer/sw_engine/*.h")
    list(APPEND TVG_SOURCES ${TVG_SW_HDR})
    add_compile_definitions(THORVG_SW_RASTER_SUPPORT)
endif()

# 3. Loader SVG
if(TVG_LOADER_SVG)
    file(GLOB_RECURSE TVG_SVG_SRC "src/loaders/svg/*.cpp")
    list(APPEND TVG_SOURCES ${TVG_SVG_SRC})
    add_compile_definitions(THORVG_SVG_LOADER_SUPPORT)
endif()

# 4. Loader Lottie
if(TVG_LOADER_LOTTIE)
    file(GLOB_RECURSE TVG_LOTTIE_SRC "src/loaders/lottie/*.cpp")
    list(APPEND TVG_SOURCES ${TVG_LOTTIE_SRC})
    add_compile_definitions(THORVG_LOTTIE_LOADER_SUPPORT)
endif()

# --- Création de la cible (Library) ---
add_library(thorvg STATIC ${TVG_SOURCES})

EASE_set_compiler_settings(thorvg "")

if(WIN32)
    target_compile_definitions(thorvg PRIVATE NOMINMAX)
    target_compile_definitions(thorvg PUBLIC -DTVG_STATIC)
endif()

target_compile_features(thorvg PUBLIC cxx_std_14)

# --- Gestion des Includes ---
target_include_directories(thorvg
    PUBLIC 
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/inc>
        $<INSTALL_INTERFACE:include>

    PRIVATE
        ${TVG_PRIVATE_INCLUDE_DIRS}
        ${CMAKE_CURRENT_BINARY_DIR} # <--- Indispensable pour trouver config.h
)

# Gestion des Threads
if(TVG_THREAD)
    find_package(Threads REQUIRED)
    target_link_libraries(thorvg PRIVATE Threads::Threads)
    target_compile_definitions(thorvg PRIVATE THORVG_THREAD_SUPPORT)
endif()

message(STATUS "ThorVG Configured (CMake Wrapper)")
